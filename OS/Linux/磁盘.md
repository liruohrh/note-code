# 分析

- 查看分区 lsblk
- 查看所有分区大小、已使用、使用率df -h
- 查看某个entry `du -sh /path`
  - 连续输出2层的 `du -xhd 2 /path`
  - `| awk -F'/' 'NF == 4'` 表示分割 / ，保留4个元素的
  - `find /path/to/dir -mindepth 1 -maxdepth 1 -print0 | du -sh --files0-from=-` 
  - `fd --print0 . /path/to/dir | du -ch --files0-from=- `
    - `fd --type f . /path/to/dir -X du -ch --`

- 查看已经删除但是未被释放的
  - `sudo lsof +L1` 筛选链接数小于 1 的文件

```bash
#!/usr/bin/env bash
# 列出已删除但仍被进程占用的文件(按大小降序)，最后给出去重后的实际占用
MIN=${1:-1048576}   # 最小字节数，默认 1 MiB

sudo lsof -nP +L1 -FpcfDisn 2>/dev/null |
awk -v min="$MIN" '
function flush() {
    if (fd != "" && size ~ /^[0-9]+$/ && size + 0 >= min + 0)
        printf "%s\t%s\t%s\t%s\t%s\t%s\n", size, pid, fd, cmd, dev ":" ino, name
    fd = size = name = dev = ino = ""
}
/^p/ { flush(); pid = substr($0, 2); cmd = ""; next }
/^c/ { cmd  = substr($0, 2); next }
/^f/ { flush(); fd = substr($0, 2); next }
/^D/ { dev  = substr($0, 2); next }
/^i/ { ino  = substr($0, 2); next }
/^s/ { size = substr($0, 2); next }
/^n/ { name = substr($0, 2); next }
END  { flush() }
' |
sort -t $'\t' -k1,1nr |
awk -F '\t' '
function human(n) {
    n += 0
    if (n >= 1073741824) return sprintf("%.2f GiB", n / 1073741824)
    if (n >= 1048576)    return sprintf("%.2f MiB", n / 1048576)
    if (n >= 1024)       return sprintf("%.2f KiB", n / 1024)
    return sprintf("%d B", n)
}
{
    printf "%12s  PID=%-7s FD=%-5s %-16s %s\n", human($1), $2, $3, $4, $6
    if (!seen[$5]++) total += $1
}
END { if (NR) printf "\n%12s  合计(同一文件只算一次，%d 条记录)\n", human(total), NR }
'
