

|      | eval      | Function |
| ---- | --------- | -------- |
| 局部变量 | ✅         | ❌        |
| 全局变量 | ✅如process | ✅        |


```js
import assert from "node:assert";

let secret = "local value";
assert.strictEqual(eval("secret"), secret);
assert.throws(
  new Function("return secret"),
  "ReferenceError: secret is not defined",
);

assert.strictEqual(eval("process"), process);
assert.strictEqual(new Function("return process")(), process);
console.log(
  JSON.stringify(
    new Function(`return Object.getOwnPropertyNames(global).toSorted()`)(),
    null,
    2,
  ),
);

```
